cmake_minimum_required(VERSION 3.18)
project(CybrForestFilm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CYBR_NATIVE "Optimize for the build CPU" OFF)
option(CYBR_SANITIZERS "Address and undefined behaviour sanitizers" OFF)
find_package(Threads REQUIRED)

# Compile the shared renderer only once, instead of rebuilding it for each CLI.
add_library(cybr-core STATIC src/scene.cpp src/integrator.cpp)
target_include_directories(cybr-core PUBLIC src)
target_link_libraries(cybr-core PUBLIC Threads::Threads)
add_library(cybr-cinema STATIC src/lightcache.cpp)
target_link_libraries(cybr-cinema PUBLIC cybr-core)
add_executable(cybr-forest src/main.cpp)
target_link_libraries(cybr-forest PRIVATE cybr-core)
add_executable(cybr-cache src/cache_main.cpp)
target_link_libraries(cybr-cache PRIVATE cybr-cinema)
add_executable(cybr-film src/film.cpp)
target_link_libraries(cybr-film PRIVATE cybr-cinema)
add_executable(cybr-bvh-reference tests/test_bvh.cpp)
target_link_libraries(cybr-bvh-reference PRIVATE cybr-core)
add_executable(cybr-cinema-tests tests/test_cinema.cpp)
target_link_libraries(cybr-cinema-tests PRIVATE cybr-cinema)

foreach(target cybr-core cybr-cinema cybr-forest cybr-cache cybr-film cybr-bvh-reference cybr-cinema-tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /W4)
  else()
    target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -Wpedantic -fno-math-errno)
    if(CYBR_NATIVE)
      target_compile_options(${target} PRIVATE -march=native)
    endif()
    if(CYBR_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME native_renderer_math_transport COMMAND cybr-forest --test)
add_test(NAME instanced_bvh_double_precision_reference COMMAND cybr-bvh-reference)
add_test(NAME cinema_cache_camera_volume COMMAND cybr-cinema-tests)
